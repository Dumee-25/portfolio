\section{LLM and AI Projects}

\project{DataSense v2: Privacy-Preserving Automated EDA}{Polars, Presidio, OpenDP}{\href{https://portfolio.kadpkumarapeli.workers.dev/systems/datasense}{portfolio case study}}
\begin{bullets}
  \item Rebuilt the NSRSIT system local-first: privacy tiers limit each prompt, Presidio finds personal data, optional
        differential privacy adds noise, and every number the model writes is checked against the facts it cites.
  \item Built a benchmark on the frozen engine: it found 4,644 of 5,421 problems planted in 50 OpenML tables, and
        27 of the 35 columns it flagged in Raha's real dirty tables had wrong cells. It still misses typos.
\end{bullets}

\project{OWL: Datacenter AI Operations Platform}{Gemini, MCP}{\href{https://github.com/Dumee-25/OWL-Datacenter_agent}{github.com/Dumee-25/OWL-Datacenter\_agent}}
\begin{bullets}
  \item Built an agentic RAG assistant over ChromaDB with hybrid vector and keyword retrieval and
        recency re-ranking, served through REST, WebSocket and a nine-tool MCP server.
  \item Paired an Isolation Forest and LSTM autoencoder ensemble with Gemini-generated root-cause reports;
        every alert carries a confidence score and the system never acts on infrastructure.
  \item Wrote a simulator producing 4.7M telemetry rows over 18 months; logged per-call LLM latency and cost.
\end{bullets}

\project{Fulyn: Private Tool-Calling Agent}{pgvector, Ollama}{\href{https://github.com/Dumee-25/Fulyn}{github.com/Dumee-25/Fulyn}}
\begin{bullets}
  \item Agent turns free-text diary entries into linked expense, sleep, mood and journal records;
        every tool call is schema-checked and invalid arguments are returned to the model to retry.
  \item Replaced vector-only search after finding unrelated queries scored 0.62 similarity, higher
        than some true matches; now merges vector and full-text rankings and flags keyword-free hits.
  \item Ran opt-in tool-calling tests across local models; llama3.2 failed most by picking the wrong tool
        or inventing arguments.
\end{bullets}
